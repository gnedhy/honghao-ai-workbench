```powershell
$releaseRoot = 'C:/ProgramData/HonghaoAI/application'

$stamp = Get-Date -Format 'yyyyMMdd-HHmmss'
Start-Process -FilePath "$releaseRoot/releases/20260915-postgresql/.venv/Scripts/python.exe" -ArgumentList '-X utf8 C:/ProgramData/HonghaoAI/application/launch-production.py' -WorkingDirectory "$releaseRoot/releases/20260915-postgresql" -WindowStyle Hidden -RedirectStandardOutput "$releaseRoot/service-$stamp.out.log" -RedirectStandardError "$releaseRoot/service-$stamp.err.log"
Invoke-RestMethod 'http://127.0.0.1:8000/api/readiness'
```
